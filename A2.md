
# Qwen3-TTS

```shell
docker run --name omni_qwen3_tts \
    --privileged \
    --shm-size=32g \
    --device /dev/davinci0 \
    --device /dev/davinci1 \
    --device /dev/davinci2 \
    --device /dev/davinci3 \
    --device /dev/davinci4 \
    --device /dev/davinci5 \
    --device /dev/davinci6 \
    --device /dev/davinci7 \
    --device /dev/davinci_manager \
    --device /dev/devmm_svm \
    --device /dev/hisi_hdc \
    -v /usr/local/dcmi:/usr/local/dcmi \
    -v /usr/local/bin/npu-smi:/usr/local/bin/npu-smi \
    -v /usr/local/Ascend/driver/lib64/:/usr/local/Ascend/driver/lib64/ \
    -v /usr/local/Ascend/driver/version.info:/usr/local/Ascend/driver/version.info \
    -v ~/.cache/:/root/.cache/ \
    -p 8091:8091 \
    -e HF_ENDPOINT=https://hf-mirror.com \
    quay.nju.edu.cn/ascend/vllm-omni:v0.24.0 \
    vllm serve Qwen/Qwen3-TTS-12Hz-0.6B-CustomVoice --omni --port 8091 --trust-remote-code
```

```shell
vllm serve Qwen/Qwen3-TTS-12Hz-0.6B-CustomVoice --omni --port 8091 --trust-remote-code

vllm bench serve --omni \
    --host 127.0.0.1 --port 8091 \
    --model Qwen/Qwen3-TTS-12Hz-0.6B-CustomVoice \
    --backend openai-audio-speech \
    --endpoint /v1/audio/speech \
    --dataset-name random \
    --random-input-len 40 --random-output-len 512 \
    --num-prompts 40 --num-warmups 1 \
    --extra-body '{"voice":"Vivian","language":"English","task_type":"CustomVoice"}' \
    --max-concurrency 5 --request-rate inf \
    --percentile-metrics ttft,e2el,audio_rtf,audio_ttfp,audio_duration,audio_underrun 
```
